# Top-level dependencies for sidewalk-auto-labeler, and the SINGLE SOURCE OF TRUTH
# for versions — environment.yml defers to this file, so pin things here only.
#
#   python -m venv .venv
#   source .venv/bin/activate        # Windows: .venv\Scripts\activate
#   pip install -r requirements.txt
#
# On linux-64 that pulls a CUDA-enabled torch wheel by default. On Windows/macOS it
# does not, so install the CUDA build first:
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu126
#
# conda works too (`conda env create -f environment.yml`) but buys nothing here: no
# project dependency comes from a conda channel, and the CUDA runtime already arrives
# as the nvidia-*-cu12 pip wheels that torch depends on.

geojson
mapbox-vector-tile  # decodes Mapillary coverage tiles (sources/mapillary.py)
numpy
opencv-python
pillow
python-dotenv  # loads MAPILLARY_ACCESS_TOKEN / PS_INTERNAL_API_KEY from ./.env
requests
scikit-image
shapely

streetlevel==0.12.10
